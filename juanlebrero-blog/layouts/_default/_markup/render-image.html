{{- /* Markdown image render hook. - Page-bundle local images: resized to 1200w WebP, lazy-loaded, explicit dimensions (no CLS). - Remote images (http/https): emitted as-is with lazy loading. */ -}} {{- $url := .Destination | safeURL -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- $isRemote := or (hasPrefix .Destination "http://") (hasPrefix .Destination "https://") (hasPrefix .Destination "//") -}} {{- if $isRemote -}} {{- else -}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- if $img -}} {{- $small := $img.Resize "1200x webp q82" -}} {{- else -}} {{- end -}} {{- end -}}
